# ============================================================
# Fabrique l'APK de « Mes accords » sur les serveurs de GitHub.
# Se lance tout seul après chaque envoi de fichiers (branche main),
# ou à la main : onglet Actions → « APK Android » → Run workflow.
# L'APK est ensuite téléchargeable dans « Releases ».
# Signature : secret CLE_ANDROID (contenu de _PRIVE_cle-android/CLE_ANDROID.txt).
# ============================================================
name: APK Android

on:
  push:
    branches: [main]
  workflow_dispatch:

permissions:
  contents: write

jobs:
  apk:
    runs-on: ubuntu-latest
    env:
      VERSION_CODE: ${{ github.run_number }}
      CLE_ANDROID: ${{ secrets.CLE_ANDROID }}
    steps:
      - name: Récupérer le projet
        uses: actions/checkout@v4

      - name: Installer Node 20
        uses: actions/setup-node@v4
        with:
          node-version: 20
          cache: npm

      - name: Installer Java 21
        uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: 21

      - name: Installer ImageMagick (icônes)
        run: sudo apt-get update -qq && sudo apt-get install -y -qq imagemagick

      - name: Construire l'appli
        run: |
          npm ci
          bash deploy/prepare-icons.sh
          npm run build
          npx cap sync android

      - name: Préparer le projet Android et votre icône
        run: |
          bash deploy/android-repair.sh
          bash deploy/android-icons.sh

      - name: Fabriquer l'APK
        run: |
          if [ -z "$CLE_ANDROID" ]; then
            echo "::warning::Secret CLE_ANDROID absent : APK de test (impossible à mettre à jour par-dessus)."
          fi
          cd android
          chmod +x gradlew
          ./gradlew --no-daemon assembleRelease
          APK=$(ls app/build/outputs/apk/release/*.apk | grep -v unsigned | head -1 || true)
          if [ -z "$APK" ]; then
            ./gradlew --no-daemon assembleDebug
            APK=$(ls app/build/outputs/apk/debug/*.apk | head -1)
          fi
          cp "$APK" "../mes-accords-${VERSION_CODE}.apk"
          ls -la ../mes-accords-*.apk

      - name: Enregistrer l'APK (Artifacts)
        uses: actions/upload-artifact@v4
        with:
          name: mes-accords-apk-${{ github.run_number }}
          path: mes-accords-*.apk

      - name: Page de téléchargement (Releases)
        uses: softprops/action-gh-release@v2
        with:
          tag_name: apk-${{ github.run_number }}
          name: Mes accords — version ${{ github.run_number }}
          make_latest: true
          body: |
            APK Android de « Mes accords ».
            Sur le téléphone : touchez le fichier .apk ci-dessous, puis « Installer ».
          files: mes-accords-*.apk
